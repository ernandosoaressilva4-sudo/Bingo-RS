const express = require('express'); const http = require('http'); const path = require('path'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server); const rooms = new Map(); function normalizeRoomCode(code = '') { return String(code).trim().toUpperCase(); } function parseNames(value = '') { return value .split(',') .map((item) => item.trim()) .filter(Boolean) .slice(0, 50); } function shuffle(array) { const copy = [...array]; for (let index = copy.length - 1; index > 0; index -= 1) { const randomIndex = Math.floor(Math.random() * (index + 1)); [copy[index], copy[randomIndex]] = [copy[randomIndex], copy[index]]; } return copy; } function buildCard(names) { if (!names.length) { return Array.from({ length: 25 }, (_, index) => `Cartela ${index + 1}`); } const list = shuffle(names); return Array.from({ length: 25 }, (_, index) => list[index % list.length]); } function getRoom(roomCode) { const normalized = normalizeRoomCode(roomCode); if (!rooms.has(normalized)) { rooms.set(normalized, { roomCode: normalized, adminSocketId: null, names: [], drawHistory: [], winners: [], clients: new Map() }); } return rooms.get(normalized); } function getPublicRoomState(room) { return { roomCode: room.roomCode, adminSocketId: room.adminSocketId, names: room.names, drawHistory: room.drawHistory, winners: room.winners, clients: Array.from(room.clients.values()).map((client) => ({ id: client.id, alias: client.alias })) }; } app.use(express.static(path.join(__dirname, 'public'))); app.get('/health', (_, response) => { response.json({ ok: true, rooms: rooms.size }); }); io.on('connection', (socket) => { socket.on('upsert-room', ({ roomCode, names }) => { const normalized = normalizeRoomCode(roomCode); const room = getRoom(normalized); const parsedNames = parseNames(names); room.names = parsedNames; room.adminSocketId = socket.id; room.drawHistory = []; room.winners = []; socket.join(room.roomCode); Array.from(room.clients.values()).forEach((client) => { client.card = buildCard(room.names); io.to(client.id).emit('my-card', { card: client.card, alias: client.alias }); }); io.to(room.roomCode).emit('room-state', getPublicRoomState(room)); }); socket.on('join-room', ({ roomCode, alias }) => { const normalized = normalizeRoomCode(roomCode); const room = getRoom(normalized); if (!room.names.length) { socket.emit('room-error', { message: 'A sala ainda não foi configurada pelo administrador.' }); return; } socket.join(room.roomCode); const cleanAlias = String(alias || '').trim() || `Cliente ${room.clients.size + 1}`; const client = { id: socket.id, alias: cleanAlias, card: buildCard(room.names) }; room.clients.set(socket.id, client); socket.emit('my-card', { card: client.card, alias: client.alias }); io.to(room.roomCode).emit('room-state', getPublicRoomState(room)); }); socket.on('draw-name', ({ roomCode }) => { const normalized = normalizeRoomCode(roomCode); const room = getRoom(normalized); if (room.adminSocketId !== socket.id) { socket.emit('room-error', { message: 'Apenas o administrador pode sortear nomes.' }); return; } if (!room.names.length) { socket.emit('room-error', { message: 'Adicione pelo menos um nome antes de sortear.' }); return; } const available = room.names.filter((name) => !room.drawHistory.includes(name)); if (!available.length) { room.drawHistory = []; } const pickFrom = room.names.filter((name) => !room.drawHistory.includes(name)); const draw = pickFrom[Math.floor(Math.random() * pickFrom.length)]; room.drawHistory = [draw, ...room.drawHistory].slice(0, 25); io.to(room.roomCode).emit('draw-result', { value: draw, history: room.drawHistory }); io.to(room.roomCode).emit('room-state', getPublicRoomState(room)); }); socket.on('declare-winner', ({ roomCode, alias }) => { const normalized = normalizeRoomCode(roomCode); const room = getRoom(normalized); if (room.adminSocketId !== socket.id) { socket.emit('room-error', { message: 'Apenas o administrador pode declarar um vencedor.' }); return; } const winnerAlias = String(alias || '').trim(); if (!winnerAlias) { return; } if (!room.winners.includes(winnerAlias)) { room.winners = [winnerAlias, ...room.winners].slice(0, 5); } io.to(room.roomCode).emit('room-state', getPublicRoomState(room)); }); socket.on('reset-room', ({ roomCode }) => { const normalized = normalizeRoomCode(roomCode); const room = getRoom(normalized); if (room.adminSocketId !== socket.id) { socket.emit('room-error', { message: 'Apenas o administrador pode resetar a sala.' }); return; } room.names = []; room.drawHistory = []; room.winners = []; room.clients.forEach((client) => { io.to(client.id).emit('my-card', { card: [], alias: client.alias }); }); room.clients.clear(); io.to(room.roomCode).emit('room-state', getPublicRoomState(room)); }); socket.on('request-room-state', ({ roomCode }) => { const room = getRoom(normalizeRoomCode(roomCode)); const client = room.clients.get(socket.id); if (client) { socket.emit('my-card', { card: client.card, alias: client.alias }); } socket.emit('room-state', getPublicRoomState(room)); }); socket.on('disconnect', () => { for (const room of rooms.values()) { if (room.adminSocketId === socket.id) { room.adminSocketId = null; } if (room.clients.has(socket.id)) { room.clients.delete(socket.id); } if (room.roomCode) { io.to(room.roomCode).emit('room-state', getPublicRoomState(room)); } } }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Servidor rodando em http://localhost:${PORT}`); }); module.exports = { app, server };
